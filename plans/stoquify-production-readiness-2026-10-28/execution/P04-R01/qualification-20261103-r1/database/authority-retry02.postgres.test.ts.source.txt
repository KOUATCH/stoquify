type Fixture = Awaited<ReturnType<typeof fixture>>

}

async function state(f: Fixture, omitRequests = false, omitActorRole = false) {
  const tables = ["organizations", "users", "roles", "sessions", "accounts", "password_history", "locations", "invites", "access_requests", "business_events", "business_event_outbox", "business_event_audits", "tenant_module_configurations", "tenant_module_grants", "tenant_module_configuration_commands"]
  const result: Record<string, { count: number; sha256: string }> = {}
  // Hash full synthetic rows, never print tokens or payloads. Auth/security audit rows are explicitly separate.
  for (const table of tables) {
    if (omitRequests && table === "access_requests") continue
    const where = table === "organizations" ? 'id = ANY($1::text[])' : ["sessions", "accounts", "password_history"].includes(table) ? '"userId" IN (SELECT id FROM users WHERE "organizationId"=ANY($1::text[]))' : '"organizationId" = ANY($1::text[])'
    let rows = await db.$queryRawUnsafe<Array<Record<string, unknown>>>(`SELECT to_jsonb(t) AS value FROM "${table}" t WHERE ${where}`, [f.org, f.foreign])
    if (omitActorRole && table === "roles") rows = rows.filter(row => (row.value as any).id !== f.actorRole.id)
    const values = rows.map(row => JSON.stringify(row.value)).sort()
    result[table] = { count: values.length, sha256: digest(values) }
  }
  const memberships = await db.$queryRawUnsafe<Array<Record<string, unknown>>>('SELECT m."A",m."B" FROM "_UserRoles" m JOIN users u ON u.id=m."B" WHERE u."organizationId"=ANY($1::text[]) ORDER BY m."A",m."B"', [f.org, f.foreign])
  result.memberships = { count: memberships.length, sha256: digest(memberships) }
  return result
}
async function deny(f: Fixture, name: string, work: () => Promise<unknown>, action = false) {
  const before = await state(f), auditBefore = await db.auditLog.count({ where: { organizationId: { in: [f.org, f.foreign] } } })
  if (action) {
    const response: any = await work()
    expect(response.success === false || response.status === 403).toBe(true)
    if (response.reason !== undefined) expect(["restricted", "session", "fresh-auth", "stale"]).toContain(response.reason)
    else { expect([401,403,404,409]).toContain(response.status); expect(response.code).not.toMatch(/INTERNAL|DATABASE|UNKNOWN/) }
    expect(response.data == null).toBe(true)
    expect(JSON.stringify(response)).not.toContain("PRIVATE_BRANCH_CANARY")
    expect(JSON.stringify(response)).not.toContain(f.session.token)
  } else await expect(work()).rejects.toMatchObject({ code: expect.stringMatching(/FORBIDDEN|FRESH_AUTH_REQUIRED|UNAUTHENTICATED|NOT_FOUND|CONFLICT|IDEMPOTENCY_CONFLICT/) })
  const after = await state(f)
  expect(after).toEqual(before)
  observations.push({ name, before, after, securityAuditRowsAdded: (await db.auditLog.count({ where: { organizationId: { in: [f.org, f.foreign] } } })) - auditBefore })
}

}

  delete payload.roleCode;delete payload.actorRoleCodes
  await db.businessEvent.create({data:{organizationId:f.org,eventType:original.eventType,eventSource:"API",schemaVersion:1,idempotencyKey:"legacy/"+legacy.id,payload,payloadHash:hashBusinessPayload(payload),actorId:f.actor,sourceId:legacy.id,occurredAt:new Date(),status:"APPLIED",processedAt:new Date()}})
  const before=await state(f), abuseBefore=await db.publicIdentityAbuseBucket.count()
  const response=await acceptInvitationWorkflow({token:legacy.token,password:"R01!"+randomBytes(24).toString("base64url")+"9a",firstName:"Synthetic",lastName:"Recipient"},{ipAddress:null})
  if(code==="admin"){
    expect(response).toMatchObject({status:403,data:null,code:"INVITATION_REISSUE_REQUIRED"})
    expect(await state(f)).toEqual(before)
    expect((await db.invite.findUniqueOrThrow({where:{id:legacy.id}})).status).toBe("PENDING")

    expect(response.status).toBe(200)
    expect((await db.invite.findUniqueOrThrow({where:{id:legacy.id}})).status).toBe("ACCEPTED")
    expect(await db.account.count({where:{userId:response.data!.id}})).toBe(1)

  }
  observations.push({name:"legacy-redemption:"+code,status:response.status,allowedAbuseBucketsAdded:(await db.publicIdentityAbuseBucket.count())-abuseBefore,businessUnchanged:code==="admin"})
})
