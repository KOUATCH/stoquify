function staticValue(node) {
  if (ts.isStringLiteral(node) || ts.isNoSubstitutionTemplateLiteral(node)) return node.text;
  if (ts.isNumericLiteral(node)) return Number(node.text);
  if (node.kind === ts.SyntaxKind.TrueKeyword) return true;
  if (node.kind === ts.SyntaxKind.FalseKeyword) return false;
  if (node.kind === ts.SyntaxKind.NullKeyword) return null;
  if (ts.isArrayLiteralExpression(node)) return node.elements.map(staticValue);
  if (ts.isObjectLiteralExpression(node)) return Object.fromEntries(node.properties.map(p => {
    assert(ts.isPropertyAssignment(p), 'nonliteral images policy requires review');
    return [p.name.text, staticValue(p.initializer)];
  }));
  throw new Error('Nonliteral image policy requires review: ' + node.kind);
}
function imagePolicy(source) {
  const sf = ts.createSourceFile('next.config.mjs', source, ts.ScriptTarget.Latest, true);
  for (const statement of sf.statements) if (ts.isVariableStatement(statement)) {
    for (const decl of statement.declarationList.declarations) if (decl.name.getText(sf) === 'nextConfig') {
      assert(ts.isObjectLiteralExpression(decl.initializer), 'nextConfig must be inspected literal');
      const images = decl.initializer.properties.find(p => ts.isPropertyAssignment(p) && p.name.getText(sf) === 'images');
      const rewrites = decl.initializer.properties.find(p => ts.isMethodDeclaration(p) && p.name.getText(sf) === 'rewrites');
      const returned = rewrites.body.statements.find(p => ts.isReturnStatement(p));
      return { images: images ? staticValue(images.initializer) : {}, rewrites: staticValue(returned.expression) };
    }
  }
  throw new Error('nextConfig images policy not found');
}

function loopbackOnly(value) {
  const address = typeof value === 'string' || value instanceof URL ? new URL(value) : value;
  const hostname = address.hostname || address.host || 'localhost';
  if (allowedPort && String(address.port || '80') !== String(allowedPort)) throw new Error('Nonfixture port blocked: '+address.port);
  if (!['localhost','127.0.0.1','::1','[::1]'].includes(hostname)) throw new Error('External network blocked: ' + hostname);
}
