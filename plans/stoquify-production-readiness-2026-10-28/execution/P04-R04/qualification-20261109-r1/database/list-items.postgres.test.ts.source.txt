async function fixture() {

  const org = prefix + "-org", foreign = prefix + "-foreign", actor = prefix + "-actor"
  await db.organization.createMany({ data: [org, foreign].map(id => ({ id, name: id, slug: id })) })
  const role = await db.role.create({ data: { organizationId: org, code: "inventory_reader", nameEn: "Inventory reader", permissions } })
  await db.user.create({ data: { id: actor, name: actor, email: actor + "@example.invalid", organizationId: org, emailVerified: true, isVerified: true, roles: { connect: { id: role.id } } } })
  const session = await db.session.create({ data: { userId: actor, token: randomUUID(), expiresAt: new Date(Date.now() + 3600000) } })
  // Actual persisted subject; expiration/signature handling of transport is not qualified.
  getSession.mockResolvedValue({ session: { id: session.id, token: session.token }, user: { id: actor, organizationId: org } })
  async function relations(organizationId: string, label: string) {
    return {
      categoryId: (await db.category.create({ data: { organizationId, titleEn: label, slug: prefix } })).id,
      brandId: (await db.brand.create({ data: { organizationId, nameEn: label, slug: prefix } })).id,
      unitId: (await db.unit.create({ data: { organizationId, nameEn: label, symbol: "ea" } })).id,
      taxRateId: (await db.taxRate.create({ data: { organizationId, nameEn: label, rate: 0 } })).id,
    }
  }
  const ownRelations = await relations(org, "Authorized reference"), foreignRelations = await relations(foreign, privateSentinel)
  const common = { organizationId: org, imageUrls: [] as string[] }
  const alpha = await db.item.create({ data: { ...common, ...ownRelations, nameEn: "Alpha fruit", nameFr: "Pomme", slug: prefix + "-a", sku: prefix + "-alpha", barcode: "111", costPrice: 10, sellingPrice: 30, createdAt: new Date("2026-11-01T00:00:01Z") } })
  const beta = await db.item.create({ data: { ...common, nameEn: "Beta fruit", nameFr: "Banane", slug: prefix + "-b", sku: prefix + "-beta", barcode: "222", costPrice: 20, sellingPrice: 10, createdAt: new Date("2026-11-01T00:00:02Z") } })
  const inactive = await db.item.create({ data: { ...common, nameEn: "Carrot", slug: prefix + "-c", sku: prefix + "-inactive", isActive: false, costPrice: 30, sellingPrice: 20, createdAt: new Date("2026-11-01T00:00:03Z") } })
  const deleted = await db.item.create({ data: { ...common, nameEn: "Deleted", slug: prefix + "-deleted", sku: prefix + "-deleted", deletedAt: new Date(), createdAt: new Date("2026-11-01T00:00:04Z") } })
  const foreignItem = await db.item.create({ data: { ...foreignRelations, organizationId: foreign, nameEn: privateSentinel, slug: prefix + "-foreign", sku: privateSentinel + prefix, imageUrls: [] } })
  return { prefix, org, foreign, actor, role, session, alpha, beta, inactive, deleted, foreignItem, ownRelations, foreignRelations }

type Fixture = Awaited<ReturnType<typeof fixture>>
const forms = ["object", "positional"] as const
function invoke(form: typeof forms[number], organizationId: string, q?: string, page?: number, pageSize?: number) {
  return form === "object" ? listItemsAction({ organizationId, q, page, pageSize }) : listItemsAction(organizationId, q, page, pageSize)
}
async function state(f: Fixture) {
  const tables = ["organizations", "users", "roles", "sessions", "items", "categories", "brands", "units", "tax_rates", "locations", "inventory_levels", "inventory_transactions", "serial_numbers", "stock_adjustments", "picture_assets", "ledger_posting_batches", "business_events", "business_event_outbox", "business_event_audits", "tenant_module_configurations", "tenant_module_grants", "tenant_module_configuration_commands"]
  const result: Record<string, { count: number; sha256: string }> = {}
  for (const table of tables) {
    const where = table === "organizations" ? 'id = ANY($1::text[])' : table === "sessions" ? '"userId" IN (SELECT id FROM users WHERE "organizationId"=ANY($1::text[]))' : table === "inventory_levels" ? '"itemId" IN (SELECT id FROM items WHERE "organizationId"=ANY($1::text[]))' : '"organizationId"=ANY($1::text[])'
    const rows = await db.$queryRawUnsafe<Array<{ value: unknown }>>(`SELECT to_jsonb(t) AS value FROM "${table}" t WHERE ${where}`, [f.org, f.foreign])
    result[table] = { count: rows.length, sha256: digest(rows.map(row => JSON.stringify(row.value)).sort()) }
  }
  for (const [name, sql] of [
    ["memberships", 'SELECT to_jsonb(t) AS value FROM "_UserRoles" t WHERE "B" IN (SELECT id FROM users WHERE "organizationId"=ANY($1::text[]))'],
    ["stock_adjustment_lines", 'SELECT to_jsonb(t) AS value FROM stock_adjustment_lines t WHERE "adjustmentId" IN (SELECT id FROM stock_adjustments WHERE "organizationId"=ANY($1::text[]))'],
  ]) {
    const rows = await db.$queryRawUnsafe<Array<{ value: unknown }>>(sql, [f.org, f.foreign])
    result[name] = { count: rows.length, sha256: digest(rows.map(row => JSON.stringify(row.value)).sort()) }
  }
  return result
}
async function observed(f: Fixture, name: string, work: () => ReturnType<typeof listItemsAction>, success: boolean, verify?: (data: any) => void) {
  const before = await state(f), auditBefore = await db.auditLog.count({ where: { organizationId: { in: [f.org, f.foreign] } } })
  const response = await work()
  expect(response.success).toBe(success)
  const serialized = JSON.stringify(response)
  expect(serialized).not.toContain(privateSentinel)
  expect(serialized).not.toContain(f.session.token)
  expect(serialized).not.toMatch(/Prisma|constraint|SELECT |UPDATE |stack|missing permission/)
  if (response.success) {
    expect(response.data.data.every(item => item.organizationId === f.org)).toBe(true)
    expect(response.data.data.map(item => item.id)).not.toContain(f.deleted.id)
    verify?.(response.data)
  } else {
    expect((response as any).data).toBeUndefined()
    expect(typeof response.error).toBe("string")
    expect(response.error.length).toBeGreaterThan(0)
  }
  const after = await state(f)
  expect(after).toEqual(before)
  expect(revalidatePath).not.toHaveBeenCalled()
  expect(revalidateTag).not.toHaveBeenCalled()
  const securityAuditRowsAdded = (await db.auditLog.count({ where: { organizationId: { in: [f.org, f.foreign] } } })) - auditBefore
  expect(securityAuditRowsAdded).toBe(success ? 0 : name.includes("revoked") ? 1 : 0)
  observations.push({ name, success: response.success, ...(response.success ? { rowIds: response.data.data.map(item => item.id), total: response.data.total, page: response.data.page, pageSize: response.data.pageSize, totalPages: response.data.totalPages } : { response }), before, after, securityAuditRowsAdded })
  return response
}
const ids = (data: any) => data.data.map((item: any) => item.id)
it.each(forms)("%s input denies revoked current database permission", async form => {
  const f = await fixture()
  await db.role.update({ where: { id: f.role.id }, data: { permissions: [] } })
  await observed(f, form + ":permission-revoked", () => invoke(form, f.org), false)
})

  const f = await fixture(); getSession.mockResolvedValue(null)
  await observed(f, form + ":unauthenticated", () => invoke(form, f.org), false)
})
it.each(forms)("%s input returns authorized rows and related DTOs", async form => {
  const f = await fixture()
  await observed(f, form + ":authorized", () => invoke(form, f.org), true, data => {
    expect(ids(data)).toEqual([f.inactive.id, f.beta.id, f.alpha.id])
    expect(data).toMatchObject({ total: 3, page: 1, pageSize: 20, totalPages: 1 })
    const alpha = data.data.find((item: any) => item.id === f.alpha.id)
    expect(alpha.category).toMatchObject({ id: f.ownRelations.categoryId, titleEn: "Authorized reference" })
    expect(alpha.brand).toMatchObject({ id: f.ownRelations.brandId, nameEn: "Authorized reference" })
    expect(alpha.unit).toMatchObject({ id: f.ownRelations.unitId, symbol: "ea" })
    expect(alpha.taxRate).toMatchObject({ id: f.ownRelations.taxRateId, nameEn: "Authorized reference" })
  })
})

it("normalizes actual Prisma query failure and restores the fixture schema", async () => {
  const f = await fixture()
  await observed(f, "database:query-fault", async () => {
    await db.$executeRawUnsafe('ALTER TABLE items RENAME COLUMN "nameEn" TO "P04R04_PRIVATE_DATABASE_DETAIL_SENTINEL"')
    try {
      await expect(db.item.findMany({ where: { organizationId: f.org } })).rejects.toMatchObject({ code: "P2022" })
      return await listItemsAction({ organizationId: f.org })
    } finally {
      await db.$executeRawUnsafe('ALTER TABLE items RENAME COLUMN "P04R04_PRIVATE_DATABASE_DETAIL_SENTINEL" TO "nameEn"')
    }
  }, false)
  expect(await db.item.count({ where: { organizationId: f.org } })).toBe(4)
})
