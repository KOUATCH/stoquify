function support(name: string, value: unknown) {
  if (!/^[a-z0-9-]+\.json$/.test(name)) throw Error("Owned private support name required")
  const file = path.resolve(privateDirectory, name)
  if (path.dirname(file) !== privateDirectory) throw Error("Private support escaped owned directory")
  const content = JSON.stringify(value, null, 2) + "\n"; fs.writeFileSync(file, content, { flag: "wx" })
  return { reference: "urn:stoquify:synthetic-monthly-close:" + name, hash: digest(content) }
}
