  const panel = "rounded-xl border border-[var(--dash-border)] bg-[var(--dash-surface)] p-4 sm:p-6", action = "h-auto min-h-10 whitespace-normal", locked = busy || !!pending
  const time = (value: string) => new Date(value).toLocaleString(locale), status = (value: string | null) => t(`statuses.${value ?? "UNVERIFIED"}`)
  const selectedInvoice = data?.invoices.find(invoice => invoice.id === selected)
  return <main className="space-y-5 text-[var(--dash-text)]" aria-busy={busy}>
    <header className="space-y-2"><Link className="text-sm text-[var(--dash-brand)] underline" href={`/${locale}/dashboard/purchases/payables`}>{t("back")}</Link><h1 className="text-2xl font-semibold sm:text-3xl">{t(kind === "request" ? "requestTitle" : "reviewTitle")}</h1><p className="max-w-3xl text-[var(--dash-text-soft)]">{t(kind === "request" ? "requestIntro" : "reviewIntro")}</p></header>
    {error && !open && <p role="alert" className={`${panel} border-[var(--dash-warning)]`}>{error}</p>}
    <section className={panel} aria-label={t("select")}><form className="mb-4 flex flex-wrap items-end gap-3" onSubmit={event=>{event.preventDefault();void load()}}><div className="min-w-0 flex-1"><Label htmlFor="exception-search">{t("search")}</Label><Input id="exception-search" value={search} onChange={event=>setSearch(event.target.value)} maxLength={120} disabled={locked}/></div><Button type="submit" variant="outline" className={action} disabled={locked||loading}>{t("refresh")}</Button></form>
      {loading&&<p role="status">{t("loading")}</p>}{!loading&&data?.invoices.length===0&&<p>{t("empty")}</p>}
      {data&&<fieldset disabled={locked||loading||open} className="space-y-2"><legend className="sr-only">{t("select")}</legend>{data.invoices.map(invoice=><label key={invoice.id} className={`flex items-start gap-3 rounded-lg border p-3 ${selected===invoice.id?'border-[var(--dash-brand)] bg-[var(--dash-brand-soft)]':'border-[var(--dash-border)]'}`}><input type="radio" name="exception-invoice" value={invoice.id} checked={selected===invoice.id} onChange={()=>choose(invoice.id)} className="mt-1 h-4 w-4 shrink-0 accent-[var(--dash-brand)]"/><span className="min-w-0 break-words"><span className="font-medium">{invoice.invoiceNumber}</span><span className="block">{invoice.total} {invoice.currency}</span>{kind==='review'&&invoice.independentReviewerRequired&&<span className="block text-sm text-[var(--dash-warning)]">{t("independent")}</span>}</span></label>)}</fieldset>}
      {selected&&<fieldset className="mt-4 grid gap-4" disabled={locked||open}><legend className="sr-only">{t("details")}</legend>{kind==='request'?<><div><Label htmlFor="exception-reason">{t("reason")}</Label><Input id="exception-reason" value={form.reason} maxLength={500} onChange={event=>setForm({...form,reason:event.target.value})}/></div><div><Label htmlFor="exception-evidence">{t("evidenceReference")}</Label><Input id="exception-evidence" value={form.evidenceReference} maxLength={500} onChange={event=>setForm({...form,evidenceReference:event.target.value})}/><p className="mt-1 text-sm text-[var(--dash-text-soft)]">{t("evidenceHelp")}</p></div><div><Label htmlFor="exception-expiry">{t("expires")}</Label><Input id="exception-expiry" type="datetime-local" value={form.expiry} onChange={event=>setForm({...form,expiry:event.target.value})}/><p className="mt-1 text-sm text-[var(--dash-text-soft)]">{t("expiryHelp")} ({Intl.DateTimeFormat().resolvedOptions().timeZone})</p></div></>:<><div><Label htmlFor="exception-record">{t("exception")}</Label><select id="exception-record" value={form.exceptionId} onChange={event=>setForm({...form,exceptionId:event.target.value})} className="mt-1 w-full rounded-md border border-[var(--dash-border)] bg-[var(--dash-surface)] p-3 text-[var(--dash-text)] focus-visible:outline focus-visible:outline-2 focus-visible:outline-[var(--dash-brand)]"><option value="">{t("chooseException")}</option>{selectedInvoice?.exceptions.map(row=><option key={row.id} value={row.id}>{row.reason} · {time(row.expiresAt)}</option>)}</select>{!selectedInvoice?.exceptions.length&&<p className="mt-2">{t("noRequests")}</p>}</div><div><Label htmlFor="exception-choice">{t("decision")}</Label><select id="exception-choice" value={form.choice} onChange={event=>setForm({...form,choice:event.target.value as "APPROVE"|"REJECT"})} className="mt-1 w-full rounded-md border border-[var(--dash-border)] bg-[var(--dash-surface)] p-3 focus-visible:outline focus-visible:outline-2 focus-visible:outline-[var(--dash-brand)]"><option value="APPROVE">{t("approve")}</option><option value="REJECT">{t("reject")}</option></select></div><div><Label htmlFor="exception-decision-reason">{t("decisionReason")}</Label><Input id="exception-decision-reason" value={form.decisionReason} maxLength={500} onChange={event=>setForm({...form,decisionReason:event.target.value})}/></div></>}</fieldset>}
      <div className="mt-4 flex flex-wrap gap-3"><Button ref={reviewButton} className={`dashboard-button-primary ${action}`} onClick={()=>void review()} disabled={locked||loading||!selected||!data||kind==='review'&&(!form.exceptionId||selectedInvoice?.independentReviewerRequired)}>{t("review")}</Button>{data?.next&&<Button variant="outline" className={action} disabled={locked||loading} onClick={()=>void load(data.next!)}>{t("more")}</Button>}</div>{data&&<p className="mt-3 text-xs text-[var(--dash-text-soft)]">{t("checked")}: {time(data.checkedAt)}</p>}
    </section>
    {selected&&<section className={panel} aria-labelledby="exception-history"><div className="flex flex-wrap items-center justify-between gap-3"><h2 id="exception-history" className="text-lg font-semibold"><History className="mr-2 inline h-5 w-5" aria-hidden="true"/>{t("history")}</h2><Button variant="outline" className={action} disabled={historyLoading||busy} onClick={()=>void loadHistory(selected)}>{t("refreshHistory")}</Button></div><p className="mt-2 text-sm">{t("historyHelp")}</p>{historyLoading&&<p role="status">{t("loading")}</p>}{historyError&&<p role="alert">{historyError}</p>}{!historyLoading&&history?.records.length===0&&<p className="mt-3">{t("historyEmpty")}</p>}<ul className="mt-3 space-y-3">{history?.records.map(row=><li key={row.recordId} className="rounded-lg border border-[var(--dash-border)] p-3"><p>{time(row.recordedAt)}</p>{row.verified&&row.decision?<><p>{t("recordedOutcome")}: {status(row.decision.status)}</p><p>{t("currentStatus")}: {status(row.observedStatus)}</p><p className="break-all text-xs">{t("request")}: {row.decision.requestReference}</p></>:<p className="text-[var(--dash-warning)]">{t("historyUnverified")}</p>}</li>)}</ul>{history?.next&&<Button variant="outline" className={`mt-3 ${action}`} disabled={historyLoading||busy} onClick={()=>void loadHistory(selected,history.next!)}>{t("moreHistory")}</Button>}</section>}
    <Dialog open={open} onOpenChange={value=>{if(!busy){setOpen(value);setPassword("");if(!value&&!pending)setPreview(null)}}}><DialogContent closeLabel={t("close")} closeDisabled={busy} onCloseAutoFocus={event=>{if(reviewButton.current&&!reviewButton.current.disabled){event.preventDefault();reviewButton.current.focus()}}} className="dashboard-landing-theme dark max-h-[90dvh] max-w-2xl overflow-y-auto text-[var(--dash-text)] motion-reduce:animate-none"><DialogHeader><DialogTitle>{t(decision?"recorded":pending?"recoverTitle":"previewTitle")}</DialogTitle><DialogDescription>{t("reviewEffect")}</DialogDescription></DialogHeader>
      {error&&<p role="alert"><AlertTriangle className="mr-2 inline h-4 w-4" aria-hidden="true"/>{error}</p>}
      {preview&&<div className="space-y-3"><p className="break-words text-lg font-medium">{preview.invoiceNumber} · {preview.supplierName}</p><p>{preview.total} {preview.currency}</p><p>{t("variance")}: {preview.variance??t("missing")} {preview.currency}</p><p className="break-words">{t("reason")}: {preview.reason}</p><p>{t("expires")}: {time(preview.expiresAt)}</p><p className="break-all">{t("evidenceReference")}: {/^(https?):\/\//i.test(preview.evidenceReference)?<a className="text-[var(--dash-brand)] underline" href={preview.evidenceReference} target="_blank" rel="noopener noreferrer">{preview.evidenceReference}</a>:preview.evidenceReference}</p>{reviewedDraft?.kind==='review'&&<p>{t("decision")}: {t(reviewedDraft.decision==='APPROVE'?'approve':'reject')} {reviewedDraft.decisionReason}</p>}<ul className="space-y-2" aria-label={t("lines")}>{preview.lines.map(line=><li key={line.id} className="rounded-lg border border-[var(--dash-border)] p-3"><p className="break-words">{line.description}</p><p>{line.quantity} × {line.unitCost} {preview.currency} · {line.total} {preview.currency}</p></li>)}</ul>{preview.blockers.map(blocker=><p key={blocker} className="text-[var(--dash-warning)]"><AlertTriangle className="mr-2 inline h-4 w-4" aria-hidden="true"/>{t(`blockers.${blocker}`)}</p>)}</div>}
      {pending&&<div className="space-y-2 text-sm"><p>{t("pending")}</p><p className="break-all">{t("request")}: {pending.idempotencyKey}</p></div>}
      {decision&&<div role="status" className="space-y-3"><p className={decision.status==='APPROVED'?'text-[var(--dash-success)]':'text-[var(--dash-text)]'}><CheckCircle2 className="mr-2 inline h-5 w-5" aria-hidden="true"/>{t(`outcomes.${decision.status}`)}</p><p>{decision.invoiceNumber} · {decision.variance} {decision.currency}</p><p>{t("decisionEffect")}</p>{decision.status==='APPROVED'&&canApproveInvoice&&<Link className="block text-[var(--dash-brand)] underline" href={`/${locale}/dashboard/purchases/payables/approve`}>{t("openApproval")}</Link>}<p>{t("recordedAt")}: {time(decision.recordedAt)}</p><p className="break-all text-xs">{t("request")}: {decision.requestReference}</p></div>}
      {!decision&&needsPassword&&<div><Label htmlFor="exception-password">{t("password")}</Label><Input id="exception-password" type="password" autoComplete="current-password" value={password} onChange={event=>setPassword(event.target.value)} disabled={busy}/><p className="mt-1 text-xs">{t("passwordHelp")}</p></div>}
      <div className="flex flex-wrap gap-3 border-t border-[var(--dash-border)] pt-3">{!decision&&(pending||preview&&!preview.blockers.length)&&<Button className={`dashboard-button-primary ${action}`} onClick={()=>void submit()} disabled={busy||needsPassword&&!password}>{busy?t("working"):t(pending?"retry":"confirm")}</Button>}<Button variant="outline" className={action} disabled={busy} onClick={()=>{setOpen(false);setPassword("");if(!pending)setPreview(null)}}>{t("close")}</Button></div>
    </DialogContent></Dialog>
  </main>
}
